# Simple Python Calculator print("--- Simple Calculator ---") print("Choose an operation: +, -, *, /") # Take inputs from the user num1 = float(input("Enter first number: ")) operator = input("Enter operator (+, -, *, /): ") num2 = float(input("Enter second number: ")) # Calculate based on the operator if operator == "+": result = num1 + num2 elif operator == "-": result = num1 - num2 elif operator == "*": result = num1 * num2 elif operator == "/": if num2 != 0: result = num1 / num2 else: result = "Error! Division by zero." else: result = "Invalid operator!" # Print the final result print(f"Result: {result}")
